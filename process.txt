//Linking 
# Install the Google Drive DVC extension
pip install "dvc[gdrive]" (on powershell)

# Link the folder ID you copied(in the main repo)
python -m dvc remote add -d myremote gdrive://1A2B3C4D5E6F...

# Push your data
python -m dvc push
